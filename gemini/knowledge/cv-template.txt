<script id="cv-data" type="application/json">
{
  "type": "cv",
  "lang": "en",
  "theme": { "color": "green", "font": "serif" },
  "basics": { "name": "Alex Morgan", "headline": "Data Analyst", "location": "Jakarta, Indonesia",
              "phone": "+62 812 0000 0000", "email": "alex.morgan@example.com",
              "links": [{ "label": "linkedin.com/in/alexmorgan", "url": "https://www.linkedin.com/in/alexmorgan" }] },
  "summary": "Data analyst with two years of experience turning sales data into dashboards and decisions.",
  "skills": [{ "group": "Analysis", "items": ["SQL", "Python (pandas)", "Excel"] },
             { "group": "Visualization", "items": ["Power BI", "Looker Studio"] }],
  "experience": [{ "title": "Northstar Retail", "subtitle": "Data Analyst", "date": "Jan 2024 – Present",
                   "bullets": ["Built Power BI sales dashboards used weekly by 40+ store managers."] }],
  "education": [{ "title": "University of Indonesia", "subtitle": "Bachelor of Statistics", "date": "2019 – 2023" }],
  "section_order": ["summary", "skills", "experience", "education"]
}
</script>

<script>
(function () {
  var ACCENTS = { green: "#4a7a62", purple: "#7b5ea7", orange: "#b86a45" };
  var FONTS = { serif: "'Source Serif 4', Georgia, 'Times New Roman', serif", sans: "Inter, Arial, sans-serif" };
  var LABELS = {
    en: { summary: "About me", skills: "Skills", experience: "Experience", projects: "Projects", education: "Education",
          certifications: "Certifications", awards: "Achievements", publications: "Publication", languages: "Languages",
          training: "Training", volunteer: "Organization" },
    id: { summary: "Tentang saya", skills: "Keahlian", experience: "Pengalaman", projects: "Proyek", education: "Pendidikan",
          certifications: "Sertifikasi", awards: "Prestasi", publications: "Publikasi", languages: "Bahasa",
          training: "Pelatihan", volunteer: "Organisasi" }
  };
  var d = JSON.parse(document.getElementById("cv-data").textContent);
  var theme = d.theme || {};
  var root = document.getElementById("root");
  document.documentElement.style.setProperty("--accent", ACCENTS[theme.color] || theme.color || ACCENTS.green);
  document.documentElement.style.setProperty("--font", FONTS[theme.font] || FONTS.serif);
  document.documentElement.lang = d.lang || "en";
  if (theme.page_size === "Letter") { var s = document.createElement("style"); s.textContent = "@page{size:Letter}"; document.head.appendChild(s); }

  function el(tag, cls, text) {
    var e = document.createElement(tag);
    if (cls) e.className = cls;
    if (text != null && text !== "") e.textContent = text;
    return e;
  }
  function add(parent, child) { if (child) parent.appendChild(child); return child; }

  var b = d.basics || {};
  document.title = (b.name || "CV") + (d.type === "cover_letter" ? " - Cover Letter" : " - CV");
  var header = add(root, el("header"));
  add(header, el("h1", "", b.name));
  if (b.headline) add(header, el("div", "headline", b.headline));
  var contact = add(header, el("div", "contact"));
  [b.location, b.phone].forEach(function (v) { if (v) add(contact, el("span", "", v)); });
  if (b.email) { var a = el("a", "", b.email); a.href = "mailto:" + b.email; add(add(contact, el("span")), a); }
  (b.links || []).forEach(function (l) { var a = el("a", "", l.label); a.href = l.url; add(add(contact, el("span")), a); });

  if (d.type === "cover_letter") {
    var L = d.letter || {};
    var box = add(root, el("div", "letter"));
    if (L.date) add(box, el("div", "date", L.date));
    var rec = add(box, el("div", "recipient"));
    (L.recipient || []).forEach(function (r) { add(rec, el("div", "", r)); });
    if (L.subject) add(box, el("div", "subject", L.subject));
    if (L.greeting) add(box, el("p", "", L.greeting));
    (L.paragraphs || []).forEach(function (p) { add(box, el("p", "", p)); });
    add(box, el("div", "", L.closing || "Sincerely,"));
    add(box, el("div", "signature", b.name));
    return;
  }

  var labels = Object.assign({}, LABELS[d.lang] || LABELS.en, d.labels || {});
  (d.section_order || ["summary", "skills", "experience", "projects", "education", "certifications", "awards"]).forEach(function (sid) {
    var label = labels[sid] || sid.replace(/_/g, " ").replace(/\b\w/g, function (c) { return c.toUpperCase(); });
    var section = el("section"), body = el("div", "body");
    section.appendChild(el("h2", "", label)); section.appendChild(body);
    if (sid === "summary") {
      if (!d.summary) return;
      add(body, el("p", "", d.summary));
    } else if (sid === "skills") {
      if (!(d.skills || []).length) return;
      var grid = add(body, el("div", "skills" + (theme.skills_layout === "single" ? " single" : "")));
      d.skills.forEach(function (g) {
        var gr = add(grid, el("div"));
        if (g.group) add(gr, el("div", "group-name", g.group));
        add(gr, el("div", "", (g.items || []).join(", ")));
      });
    } else {
      var items = d[sid];
      if (!Array.isArray(items) || !items.length) return;
      items.forEach(function (e) {
        var it = add(body, el("div", "item"));
        var h = add(it, el("div", "item-head"));
        add(h, el("div", "item-title", e.title));
        if (e.date) add(h, el("div", "item-meta", e.date));
        if (e.subtitle || e.location) {
          var h2 = add(it, el("div", "item-head"));
          add(h2, el("div", "item-sub", e.subtitle || ""));
          if (e.location) add(h2, el("div", "item-meta", e.location));
        }
        if (e.description) add(it, el("div", "", e.description));
        if ((e.bullets || []).length) {
          var ul = add(it, el("ul"));
          e.bullets.forEach(function (t) { add(ul, el("li", "", t)); });
        }
      });
    }
    root.appendChild(section);
  });
})();
</script>
